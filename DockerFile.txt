# 1. تحديد النسخة الأساسية من بايثون
FROM python:3.10-slim

# 2. تحديد مجلد العمل داخل الـ Container
WORKDIR /app

# 3. نسخ ملف المتطلبات أولاً لتسريع عملية البناء (Caching)
COPY requirements.txt .

# 4. تثبيت المكتبات المطلوبة بدون الاحتفاظ بالكاش لتقليل حجم الصورة
RUN pip install --no-cache-dir -r requirements.txt

# 5. نسخ كافة ملفات المشروع إلى مجلد العمل
COPY . .

# 6. التنويه بأن التطبيق سيعمل على المنفذ 8000
EXPOSE 8000

# 7. الأمر النهائي لتشغيل Chainlit وتحديد العنوان والمنفذ المناسب للـ Render
CMD ["chainlit", "run", "app.py", "--host", "0.0.0.0", "--port", "8000"]
